function doGet(e) { try { var callback = e.parameter.callback; var datosStr = e.parameter.datos; if (datosStr) { var datos = JSON.parse(decodeURIComponent(datosStr)); var url = generarDocumento(datos); var response = JSON.stringify({ok: true, url: url}); if (callback) { return ContentService .createTextOutput(callback + '(' + response + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(response) .setMimeType(ContentService.MimeType.JSON); } return ContentService .createTextOutput(JSON.stringify({ok: true, mensaje: 'Web App Cocuzza activa'})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { var errResponse = JSON.stringify({ok: false, error: err.toString()}); var cb = e.parameter.callback; if (cb) { return ContentService .createTextOutput(cb + '(' + errResponse + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(errResponse) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var datos = JSON.parse(e.postData.contents); var url = generarDocumento(datos); return ContentService .createTextOutput(JSON.stringify({ok: true, url: url})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ok: false, error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function generarDocumento(d) { var hoy = new Date(); var dia = hoy.getDate(); var meses = ['enero','febrero','marzo','abril','mayo','junio','julio','agosto','septiembre','octubre','noviembre','diciembre']; var mes = meses[hoy.getMonth()]; var anio = hoy.getFullYear(); var fechaLarga = dia + ' días del mes de ' + mes + ' de ' + anio; var fechaCorta = dia + ' de ' + mes + ' de ' + anio; var carpeta = DriveApp.getFolderById(FOLDER_ID); var template = DriveApp.getFileById(TEMPLATE_ID); var nombreArchivo = 'Reserva_' + (d.nombre || 'SinNombre').replace(/[^a-zA-ZáéíóúÁÉÍÓÚñÑ\s]/g,'').trim().replace(/\s+/g,'_') + '_' + dia + mes.substring(0,3) + anio; var newFile = template.makeCopy(nombreArchivo, carpeta); var doc = DocumentApp.openById(newFile.getId()); var body = doc.getBody(); var vars = { '{{FECHA}}': fechaLarga, '{{FECHA_CORTA}}': fechaCorta, '{{NOMBRE_COMPRADOR}}': d.nombre || '', '{{DNI}}': d.dni || '', '{{CELULAR}}': d.celular || '', '{{EMAIL}}': d.email || '', '{{DOMICILIO_COMPRADOR}}': d.domicilio || '', '{{CIUDAD_COMPRADOR}}': d.ciudadComp || '', '{{MONTO_LETRAS}}': d.montoReservaLetras || '', '{{SIMBOLO}}': d.simbolo || '$', '{{MONTO_NUMEROS}}': d.montoReservaNum || '', '{{CALLE_INMUEBLE}}': d.calleInmueble || '', '{{ENTRE_CALLES}}': d.entreCalles || '', '{{BARRIO}}': d.barrio || 'Centro', '{{CIUDAD_INMUEBLE}}': d.ciudadInmueble || '', '{{DESCRIPCION}}': d.descripcion || '', '{{PRECIO_LETRAS}}': d.precioTotalLetras || '', '{{PRECIO_NUMEROS}}': d.precioTotalNum || '', '{{BOLETO_LETRAS}}': d.montoBoletoLetras || '', '{{BOLETO_NUMEROS}}': d.montoBoletoNum || '', '{{FORMA_PAGO}}': d.formaPago || '', '{{TOTAL_CONTADO}}': d.totalContado || '-', '{{HONORARIOS}}': d.honorarios || '-', '{{SELLADOS}}': d.sellados || '-', '{{INFORMES}}': d.informes || '-', '{{OTROS}}': d.otros || '-', '{{TOTAL}}': d.total || '-', '{{RECIBIDO}}': d.recibido || '-', '{{SALDO}}': d.saldo || '-' }; for (var key in vars) { body.replaceText(key, vars[key]); } doc.saveAndClose(); Logger.log('Reserva generada: ' + newFile.getUrl()); return newFile.getUrl(); }
